from app.models import Material

# AJOUTER
new_material = Material(code='NEW001', name='Nouveau Materiel', unit='piece')

db.session.commit()  # Sauvegarde en base

# LIRE

# MODIFIER

db.session.commit()  # Sauvegarde la modification

# VERIFIER
updated = Material.query.get(material.id)
print(updated.current_stock)  # Affiche 100

# SUPPRIMER
db.session.delete(material)
db.session.commit()  # Sauvegarde la suppression
